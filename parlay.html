document.addEventListener('click', e => { const pf = e.target.closest('.pick-fighter'); if (pf) { const fightId = parseInt(pf.dataset.fight, 10); const fighterId = parseInt(pf.dataset.fighter, 10); if (!fighterId) return; // Toggle: if already selected, remove; if other side selected, replace const existingIndex = legs.findIndex(l => l.fightId === fightId); const wasAdded = !(existingIndex >= 0 && legs[existingIndex].fighterId === fighterId); if (existingIndex >= 0 && legs[existingIndex].fighterId === fighterId) { legs.splice(existingIndex, 1); } else { const newLeg = { fightId, fighterId, fighterName: pf.dataset.name, opponentName: pf.dataset.opp, odds: '' }; if (existingIndex >= 0) { legs[existingIndex] = newLeg; } else { legs.push(newLeg); } } render(); renderSlip(); if (wasAdded) pulseSlip(); } });
